@P.autodetect_parameters()
@P.case(
    name="Nil",
    clist=Nil(),
    pylist=[],
    expected=True,
)
@P.case(
    name="single",
    clist=Cons.pure(1),
    pylist=[1],
    expected=True,
)
@P.case(
    name="multiple",
    clist=2 << Cons(1),
    pylist=[2,1],
    expected=True
)
@P.case(
    name="nested",
    lst=Cons(4, Cons(Cons(3, Cons(2, Nil())), Cons(5, Nil()))),
    expected="CList([4, [3, 2], 5])",
)
def test_cons_list_can_equal_py_list(lst, expected):
    actual = str(lst)
    assert actual == expected
